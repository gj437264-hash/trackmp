1. The Problem

The original /api/admin/politicians/duplicates route ran a live MongoDB aggregation (grouping by normalized name) plus an O(n²) Python pairwise comparison inside every name-group — all synchronously inside the HTTP request/response cycle.

With 500,000+ politician rows this exceeded nginx's proxy_read_timeout, producing a 504 Gateway Timeout. It would only get worse heading toward 2M+ rows.

Root cause: two expensive operations stacked on one request — a full-collection aggregation with no index on the grouping field, and unbounded O(n²) comparison work, both blocking the request thread until nginx gave up.

2. The Solution: Offline Precompute

Move the expensive computation out of the request cycle entirely — same pattern already used for TrackMP's SEO pipeline (seo_tasks + systemd timers + atomic writes):

PieceRole
standalone_compute_duplicate_v2.pyRuns the aggregation + matching offline, writes results to a duplicate_candidates collection
duplicate_finder.py (route)Becomes a trivial paginated read against the precomputed collection — fast at any scale
systemd timerReruns the script nightly (or on demand) so results stay fresh

Also fixed: grouping key changed from name alone to (name, country_code), shrinking group sizes and cutting the O(n²) workload — cross-country name collisions usually aren't real duplicates anyway.

3. Step 1 — Add the Compound Index

Do this once, regardless of everything else below:

mongosh
use trackmp
db.politicians.createIndex({ name: 1, country_code: 1 })

4. Step 2 — Standalone Compute Script

FILE: /opt/trackmp/backend/standalone_compute_duplicate_v2.py

Run manually or via timer. Bypasses nginx/FastAPI entirely — no request timeout to hit. Includes an n > 300 safety valve so a pathologically large name/country group gets flagged for manual review instead of hanging the whole run, and writes results via an atomic rename so the live route never sees a half-written table.

"""
Standalone offline job: computes duplicate politician clusters and
writes them into `duplicate_candidates`. Run manually or via systemd timer.
Bypasses nginx/FastAPI entirely, so there's no request timeout to hit.

Usage:
    python standalone_compute_duplicate_v2.py [--country US] [--min-matches 1]
"""
from __future__ import annotations

import argparse
import asyncio
import logging
import os
from datetime import datetime, timezone
from pathlib import Path

from dotenv import load_dotenv
from motor.motor_asyncio import AsyncIOMotorClient

# Load .env explicitly by path so this works no matter what directory
# you run the script from.
load_dotenv(Path(__file__).resolve().parent / ".env")

MONGO_URL = os.environ["MONGO_URL"]
DB_NAME = os.environ["DB_NAME"]

logging.basicConfig(level=logging.INFO, format="%(asctime)s [%(levelname)s] %(message)s")
log = logging.getLogger("compute_duplicates")

_GEO_FIELDS = ["country_code", "state_id", "city_id", "constituency_id"]
_BATCH_SIZE = 500


class _UnionFind:
    def __init__(self, n: int):
        self.parent = list(range(n))

    def find(self, x: int) -> int:
        while self.parent[x] != x:
            x = self.parent[x]
        return x

    def union(self, a: int, b: int) -> None:
        ra, rb = self.find(a), self.find(b)
        if ra != rb:
            self.parent[rb] = ra


def _pairwise_match(a: dict, b: dict) -> list[str]:
    matched = []
    for field in _GEO_FIELDS:
        va, vb = a.get(field), b.get(field)
        if va and vb and va == vb:
            matched.append(field)
    return matched


async def compute(db, country_code: str | None, min_matches: int) -> list[dict]:
    match_stage: dict = {"deleted": {"$ne": True}}  # adjust to match your NOT_DELETED filter exactly
    if country_code:
        match_stage["country_code"] = country_code.strip().upper()

    pipeline = [
        {"$match": match_stage},
        {"$addFields": {
            "_norm_name": {"$trim": {"input": {"$toLower": {"$ifNull": ["$name", ""]}}}},
        }},
        {"$match": {"_norm_name": {"$ne": ""}}},
        {"$group": {
            "_id": {"name": "$_norm_name", "country_code": "$country_code"},
            "count": {"$sum": 1},
            "docs": {"$push": {
                "id": {"$toString": "$_id"},
                "name": "$name",
                "party": "$party",
                "role": "$role",
                "country_code": "$country_code",
                "state_id": "$state_id",
                "city_id": "$city_id",
                "constituency_id": "$constituency_id",
                "verified": "$verified",
                "image_url": "$image_url",
            }},
        }},
        {"$match": {"count": {"$gte": 2}}},
    ]

    log.info("Running duplicate aggregation (country=%s, min_matches=%s)...",
              country_code or "ALL", min_matches)
    cursor = db.politicians.aggregate(pipeline, allowDiskUse=True, batchSize=200)

    clusters = []
    groups_seen = 0
    async for group in cursor:
        groups_seen += 1
        docs = group["docs"]
        n = len(docs)

        if n > 300:
            log.warning("Skipping O(n^2) match for oversized group %r (%d records) — "
                        "flagging for manual review instead.", group["_id"], n)
            clusters.append({
                "name": docs[0]["name"],
                "country_code": group["_id"]["country_code"],
                "matched_fields": ["__manual_review__"],
                "score": 1,
                "members": docs,
            })
            continue

        uf = _UnionFind(n)
        pair_reasons: dict[tuple, list[str]] = {}
        for i in range(n):
            for j in range(i + 1, n):
                matched = _pairwise_match(docs[i], docs[j])
                if len(matched) >= min_matches:
                    uf.union(i, j)
                    pair_reasons[(i, j)] = matched

        cluster_members: dict[int, list[int]] = {}
        for i in range(n):
            cluster_members.setdefault(uf.find(i), []).append(i)

        for member_indexes in cluster_members.values():
            if len(member_indexes) < 2:
                continue
            reasons: set[str] = set()
            for (i, j), fields in pair_reasons.items():
                if i in member_indexes and j in member_indexes:
                    reasons.update(fields)
            clusters.append({
                "name": docs[member_indexes[0]]["name"],
                "country_code": group["_id"]["country_code"],
                "matched_fields": sorted(reasons),
                "score": 1 + len(reasons),
                "members": [docs[i] for i in member_indexes],
            })

        if groups_seen % _BATCH_SIZE == 0:
            log.info("Processed %d name-groups so far...", groups_seen)
            await asyncio.sleep(0)

    clusters.sort(key=lambda c: (-c["score"], (c["name"] or "").lower()))
    log.info("Done. %d name-groups scanned, %d duplicate clusters found.", groups_seen, len(clusters))
    return clusters


async def resolve_geo_names(db, clusters: list[dict]) -> None:
    from bson import ObjectId
    from bson.errors import InvalidId

    def safe_oid(id_):
        if not id_:
            return None
        try:
            return ObjectId(id_)
        except (InvalidId, TypeError):
            return None

    state_cache, city_cache, const_cache = {}, {}, {}
    for cluster in clusters:
        for m in cluster["members"]:
            sid, cid, kid = m.get("state_id"), m.get("city_id"), m.get("constituency_id")
            if sid and sid not in state_cache:
                oid = safe_oid(sid)
                doc = await db.states.find_one({"_id": oid}) if oid else None
                state_cache[sid] = doc.get("name") if doc else None
            if cid and cid not in city_cache:
                oid = safe_oid(cid)
                doc = await db.cities.find_one({"_id": oid}) if oid else None
                city_cache[cid] = doc.get("name") if doc else None
            if kid and kid not in const_cache:
                oid = safe_oid(kid)
                doc = await db.constituencies.find_one({"_id": oid}) if oid else None
                const_cache[kid] = doc.get("name") if doc else None
            m["state_name"] = state_cache.get(sid)
            m["city_name"] = city_cache.get(cid)
            m["constituency_name"] = const_cache.get(kid)


async def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--country", default=None)
    parser.add_argument("--min-matches", type=int, default=1)
    args = parser.parse_args()

    client = AsyncIOMotorClient(MONGO_URL)
    db = client[DB_NAME]

    try:
        clusters = await compute(db, args.country, args.min_matches)
        await resolve_geo_names(db, clusters)

        now = datetime.now(timezone.utc)
        for c in clusters:
            c["computed_at"] = now

        tmp_name = "duplicate_candidates_tmp"
        await db[tmp_name].drop()
        if clusters:
            await db[tmp_name].insert_many(clusters)
        await db[tmp_name].create_index([("score", -1), ("name", 1)])
        await db[tmp_name].create_index("country_code")

        # renameCollection must be issued against the admin database,
        # regardless of which DB the collections live in.
        await client.admin.command({
            "renameCollection": f"{DB_NAME}.{tmp_name}",
            "to": f"{DB_NAME}.duplicate_candidates",
            "dropTarget": True,
        })
        log.info("Wrote %d clusters to duplicate_candidates.", len(clusters))
    finally:
        client.close()


if __name__ == "__main__":
    asyncio.run(main())

5. Fixing the ModuleNotFoundError

ISSUE HIT DURING DEPLOY

ModuleNotFoundError: No module named 'config' — the script originally guessed at a config module with MONGO_URI/MONGO_DB_NAME. TrackMP's actual .env uses MONGO_URL and DB_NAME instead. Fixed by loading .env directly with python-dotenv (already reflected in the script above).

# confirm python-dotenv is installed
pip show python-dotenv || pip install python-dotenv

# script must live at backend root, not inside addon_routes/
mv /opt/trackmp/backend/addon_routes/standalone_compute_duplicate.py \
   /opt/trackmp/backend/standalone_compute_duplicate_v2.py
addon_routes/ is for FastAPI routers only — this is a standalone CLI script and belongs at backend/ root, next to server.py.

6. Fixing the renameCollection Error

ISSUE HIT DURING DEPLOY

pymongo.errors.OperationFailure: renameCollection may only be run against the admin database.

The aggregation and write succeeded fully (162,049 clusters computed) — only the final atomic rename failed, because MongoDB requires renameCollection to be issued against the admin database specifically, not whichever database the collections live in.

BeforeAfter
await db.command({"renameCollection": ...}) await client.admin.command({"renameCollection": ..., "dropTarget": True})

Added "dropTarget": True too, so the rename atomically replaces the old duplicate_candidates in one step (no gap where the collection doesn't exist), letting you drop the separate .drop() call beforehand.

7. Step 3 — Replacement Route (Cache Reader)

FILE: /opt/trackmp/backend/addon_routes/duplicate_finder.py

Fully replaces the old live-aggregation route. Now a plain indexed find() — sub-millisecond regardless of how large politicians grows.

"""
Duplicate politician detection — reads from precomputed results.

The actual duplicate-matching (cascading name -> country -> state -> city
-> constituency comparison) now runs OFFLINE via
`standalone_compute_duplicate_v2.py`, which writes its results into the
`duplicate_candidates` collection. This route just serves that precomputed
data with pagination, so it stays fast (indexed find, no aggregation)
regardless of how large the `politicians` collection grows.

To refresh the results, run the standalone script manually or via its
systemd timer:
    /opt/trackmp/backend/venv/bin/python3 standalone_compute_duplicate_v2.py

This endpoint is READ-ONLY: it never modifies any politician record. It
is gated behind the same `require_section("politicians")` dependency used
by every other admin politician route, so access control matches the
rest of the admin panel.
"""
from __future__ import annotations

from typing import Optional

from fastapi import APIRouter, Depends, Query

from routes import get_db, require_section

router = APIRouter(prefix="/api")

_MAX_LIMIT = 100


@router.get("/admin/politicians/duplicates")
async def find_duplicate_politicians(
    country_code: Optional[str] = Query(
        None,
        max_length=2,
        description="Restrict results to one country.",
    ),
    page: int = Query(1, ge=1),
    limit: int = Query(20, ge=1, le=_MAX_LIMIT),
    user: dict = Depends(require_section("politicians")),
):
    db = get_db()

    match: dict = {}
    if country_code:
        match["country_code"] = country_code.strip().upper()

    total = await db.duplicate_candidates.count_documents(match)
    cursor = (
        db.duplicate_candidates
        .find(match, {"_id": 0})
        .sort([("score", -1), ("name", 1)])
        .skip((page - 1) * limit)
        .limit(limit)
    )
    items = await cursor.to_list(limit)
    computed_at = items[0]["computed_at"] if items else None

    return {
        "items": items,
        "total": total,
        "page": page,
        "limit": limit,
        "computed_at": computed_at,
    }
Removed vs. the original: _UnionFind, _pairwise_match, _safe_oid, _GEO_FIELDS, the live aggregation pipeline, and the min_matches query param (matching threshold is now baked in at script-run time). If the frontend was sending min_matches, FastAPI won't error — it'll just be silently ignored. Check: grep -rn "min_matches" /opt/trackmp/frontend/src/

[Unit]
Description=Recompute politician duplicate candidates

[Service]
Type=oneshot
WorkingDirectory=/opt/trackmp/backend
ExecStart=/opt/trackmp/backend/venv/bin/python3 standalone_compute_duplicate_v2.py
User=www-data
FILE: /etc/systemd/system/duplicate-scan.timer 
[Unit]
Description=Nightly duplicate scan

[Timer]
OnCalendar=*-*-* 03:00:00
Persistent=true

[Install]
WantedBy=timers.target
systemctl daemon-reload
systemctl enable --now duplicate-scan.timer
systemctl list-timers | grep duplicate-scan

To run it on demand at any time (e.g. right after a data cleanup pass):

sudo -u www-data /opt/trackmp/backend/venv/bin/python3 \

10. Sanity-Checking Results

A very high cluster count (e.g. 162,049) is worth spot-checking before trusting it on the dashboard — with --min-matches 1, common names sharing just a country will cluster even if they're different people. Bump to --min-matches 2 for stricter matching.

sudo -u www-data /opt/trackmp/backend/venv/bin/python3 - <<'EOF'
import asyncio
from motor.motor_asyncio import AsyncIOMotorClient
import os
from dotenv import load_dotenv
load_dotenv("/opt/trackmp/backend/.env")

async def main():
    client = AsyncIOMotorClient(os.environ["MONGO_URL"])
    db = client[os.environ["DB_NAME"]]
    total = await db.duplicate_candidates.count_documents({})
    print("total clusters:", total)
    async for c in db.duplicate_candidates.find({}, {"_id": 0}).sort("score", -1).limit(5):
        print(c["name"], c["country_code"], c["matched_fields"], "members:", len(c["members"]))
    client.close()

asyncio.run(main())
EOF
